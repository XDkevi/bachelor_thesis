WITH filtered AS (
    SELECT
        V2Themes
    FROM `gdelt-bq.gdeltv2.gkg_partitioned`
    WHERE
        _PARTITIONTIME BETWEEN TIMESTAMP('2024-01-01') AND TIMESTAMP('2025-12-31')
        AND DATE BETWEEN 20240101000000 AND 20251231235959
        AND TranslationInfo IS NULL
        AND SourceCommonName IN (
            'reuters.com', 'bloomberg.com', 'ft.com', 'wsj.com', 'cnbc.com', 'economist.com'
        )
        AND NOT REGEXP_CONTAINS(
            LOWER(DocumentIdentifier),
            r'(/fashion|/sports|/entertainment|/lifestyle|/celebrity|/culture|/travel|/luxury|/video|podcast|newsletter|live-blog|liveblog|morning-briefing|market-wrap)'
        )
        AND REGEXP_CONTAINS(
            UPPER(V2Locations),
            r'(US|EZ|GM|FR|IT|UKRAINE|RUSSIA|CHINA|IRAN|ISRAEL|TAIWAN)'
        )
        AND ABS(CAST(SPLIT(V2Tone, ',')[OFFSET(0)] AS FLOAT64)) >= 2
        
        AND V2Themes IS NOT NULL
),

expanded AS (
    SELECT
        SPLIT(TRIM(raw_theme), ',')[OFFSET(0)] AS Theme
    FROM filtered,
    UNNEST(SPLIT(V2Themes, ';')) AS raw_theme
    WHERE raw_theme != ''
),

macro_themes AS (
    SELECT Theme
    FROM expanded
    WHERE REGEXP_CONTAINS(
        UPPER(Theme),
        r'^(ECON_|WB_|EPU_|ENV_|FINANCIAL_|HEALTH_|ARMED|WAR|MILITARY|UNEMPLOYMENT|INFLATION|SOVEREIGN|TAX_ECON|TAX_DISEASE)'
    )
)

SELECT
    Theme,
    COUNT(*) AS Occurrences
FROM macro_themes
GROUP BY Theme
ORDER BY Occurrences DESC
LIMIT 200;

        14400 * DIV(UNIX_SECONDS(PARSE_TIMESTAMP('%Y%m%d%H%M%S', CAST(DATE AS STRING))), 14400)

        INTERVAL 4 HOUR

    _PARTITIONTIME BETWEEN TIMESTAMP('2024-01-01') AND TIMESTAMP('2024-12-31')
    AND DATE BETWEEN 20240101000000 AND 20241231235959

        REGEXP_CONTAINS(

scored AS (
  SELECT *,
    CASE
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ECON_CENTRALBANK|WB_1235_CENTRAL_BANKS|EPU_POLICY_CENTRAL_BANK|EPU_POLICY_FEDERAL_RESERVE)') THEN 100
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ECON_INFLATION|WB_442_INFLATION)') THEN 95
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ECON_INTEREST_RATES|EPU_CATS_MONETARY_POLICY|EPU_POLICY_MONETARY_POLICY|EPU_POLICY_INTEREST_RATES?|WB_444_MONETARY_POLICY)') THEN 90
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ECON_DEBT|WB_450_DEBT|WB_1104_MACROECONOMIC_VULNERABILITY_AND_DEBT|SOVEREIGN_DEBT)') THEN 89
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(UNEMPLOYMENT|WB_471_ECONOMIC_GROWTH|WB_2670_JOBS|EPU_ECONOMY|EPU_ECONOMY_HISTORIC|WB_439_MACROECONOMIC_AND_STRUCTURAL_POLICIES)') THEN 88
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(FINANCIAL_CRISIS|WB_318_FINANCIAL_ARCHITECTURE_AND_BANKING|WB_1234_BANKING_INSTITUTIONS|WB_1920_FINANCIAL_SECTOR_DEVELOPMENT|ECON_BANKRUPTCY)') THEN 85
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ENV_ENERGY|ENV_OIL|ECON_OILPRICE|ENV_NATURALGAS)') THEN 80
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(ECON_SANCTIONS|ECON_TRADE_DISPUTE|ECON_TARIFFS|WB_698_TRADE|EPU_CATS_TRADE_POLICY)') THEN 75
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(WAR|ARMEDCONFLICT|MILITARY)') THEN 70
      WHEN REGEXP_CONTAINS(UPPER(V2Themes), r'(HEALTH_PANDEMIC|WB_2167_PANDEMICS|TAX_DISEASE_CORONAVIRUS|TAX_DISEASE_COVID)') THEN 65
      ELSE 50
    END AS MacroPriority
  FROM filtered
)

SELECT *
FROM scored
QUALIFY
  ROW_NUMBER() OVER (
    PARTITION BY Candle4H
    ORDER BY 
      MacroPriority DESC,
      ABS(SentimentScore) DESC,
      PublishTimestamp DESC
  ) <= 20
ORDER BY PublishTimestamp;
